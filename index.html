async function loadRadarResults() { const status = document.getElementById("radar-status"); const updated = document.getElementById("radar-updated"); const tbody = document.getElementById("radar-results"); try { // Evita reutilizar una respuesta antigua de la caché. const url = "./data/new-pools.json?t=" + Date.now(); const response = await fetch(url, { cache: "no-store" }); if (!response.ok) { throw new Error("HTTP " + response.status); } const result = await response.json(); const pools = result.pools || []; status.textContent = pools.length + " pares encontrados en la última búsqueda."; updated.textContent = result.updated_at ? "Última actualización UTC: " + result.updated_at : ""; tbody.replaceChildren(); if (pools.length === 0) { const tr = document.createElement("tr"); const td = document.createElement("td"); td.colSpan = 6; td.textContent = "No se encontraron pares en esta consulta."; tr.appendChild(td); tbody.appendChild(tr); return; } for (const pool of pools) { const tr = document.createElement("tr"); const values = [ pool.name || "Par sin nombre", pool.price_usd == null ? "N/D" : "$" + pool.price_usd, pool.liquidity_usd == null ? "N/D" : "$" + Number(pool.liquidity_usd).toLocaleString(), pool.volume_24h_usd == null ? "N/D" : "$" + Number(pool.volume_24h_usd).toLocaleString(), pool.price_change_24h == null ? "N/D" : pool.price_change_24h + "%" ]; for (const value of values) { const td = document.createElement("td"); td.textContent = value; tr.appendChild(td); } const linkCell = document.createElement("td"); const link = document.createElement("a"); // Solo se permiten enlaces a la web oficial de GeckoTerminal. if (pool.address && /^[a-zA-Z0-9]+$/.test(pool.address)) { link.href = "https://www.geckoterminal.com/base/pools/" + encodeURIComponent(pool.address); link.target = "_blank"; link.rel = "noopener noreferrer"; link.textContent = "Ver par"; } else { link.textContent = "No disponible"; } linkCell.appendChild(link); tr.appendChild(linkCell); tbody.appendChild(tr); } } catch (error) { status.textContent = "No se pudieron cargar los resultados del escáner."; updated.textContent = "Comprueba que GitHub Pages ya publicó data/new-pools.json."; console.error("Error al cargar CryptoRadar:", error); } } loadRadarResults(); // Actualiza la vista cada 5 minutos mientras la página está abierta. setInterval(loadRadarResults, 5 * 60 * 1000);
